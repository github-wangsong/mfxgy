# 02 · 真实模型调用与流式（LLM 节点）

> 阶段 5 · 第 2 步
>
> 做完：`llm` 节点从「Echo 回显」换成真实 provider 调用；`token` 事件来自模型逐块输出；`node_finished.output` 带上真实 `model` 与 `usage`；`model` 为空串时回退到 `LLM_MODEL`；没有密钥时立刻失败而不是发出半截流。
>
> **本篇面试考点**：`stream()` 为什么要 `await` 两次 · `maxRetries: 0` 的两端一致性 · `chunk.content` 的三种形态 · `usage` 为什么在最后一帧 · 不传 `temperature` vs 传 `undefined` · 为什么 `text` 只能由服务端累积 · 配置读取与客户端创建的位置 · 流式调用的超时设置。

## 1. 与阶段 4 的差异

| 项 | 阶段 4（占位） | 阶段 5（真实） |
| --- | --- | --- |
| 文本来源 | `Echo: <渲染后的 prompt>` | provider 逐块返回 |
| `token` 帧数 | 1 帧 | 每个非空 chunk 一帧 |
| `model` | 固定 `'stub'` | 实际模型名（回退后） |
| `usage` | 无 | provider 返回时才有 |
| 失败方式 | 模板缺变量 | 模板缺变量 + 认证/限流/超时/5xx |
| 依赖 | 无 | `@langchain/openai` / `langchain-openai` |

引擎与前端**不用改**：`runLlmNode` 的签名、输出形状、事件语义都没变。这就是阶段 4 坚持“占位也走完整协议”的回报。

## 2. TypeScript 实现

**完整文件路径：`apps/api-ts/src/engine/nodes/llm.node.ts`**（阶段 5 版本）

```ts
import { HumanMessage, SystemMessage, type BaseMessage, type MessageContent } from '@langchain/core/messages'
import { ChatOpenAI } from '@langchain/openai'
import type { GraphNode, LlmNodeData } from '@mini-dify/contract'

import { RunError } from '../events'
import { renderTemplate, type Pool } from '../template'

/** 一轮 LLM 调用的结果（阶段 5 无工具，字段刻意保持与后续阶段兼容）。 */
export interface LlmStepResult {
  /** 拼接后的完整文本 */
  text: string
  /** 实际使用的模型名（回退后） */
  model: string
  /** provider 提供时才有 */
  usage?: { promptTokens: number; completionTokens: number; totalTokens: number }
}

/**
 * 构造 `llm` 节点的消息数组。
 * `system` 非空才加；`prompt` 渲染后作为最后一条 `HumanMessage`。
 */
export function buildMessages(data: LlmNodeData, pool: Pool): BaseMessage[] {
  const messages: BaseMessage[] = []
  if (typeof data.system === 'string' && data.system !== '') {
    messages.push(new SystemMessage(data.system))
  }
  messages.push(new HumanMessage(renderTemplate(data.prompt, pool)))
  return messages
}

/**
 * 执行一轮流式 LLM 调用。
 *
 * - 每个非空 chunk 立即 `emitToken`（前端据此逐字渲染）；
 * - 同时把片段累积到 `text`（服务端是全文的唯一真源）；
 * - 任一异常交给 `mapLlmError` 归一为规范文案（见 03 篇）。
 */
export async function runLlmStep(
  data: LlmNodeData,
  messages: BaseMessage[],
  emitToken: (text: string) => void,
): Promise<LlmStepResult> {
  // 1. 没有密钥：立刻失败，不创建客户端、不发任何 token
  const apiKey = process.env.LLM_API_KEY ?? ''
  if (apiKey === '') {
    throw new RunError('llm provider not configured: LLM_API_KEY')
  }

  // 2. model 回退：节点为空串 → 部署默认值
  const model = data.model ? data.model : (process.env.LLM_MODEL ?? '')
  const client = new ChatOpenAI({
    apiKey,
    model,
    configuration: { baseURL: process.env.LLM_BASE_URL ?? 'https://api.openai.com/v1' },
    timeout: Number(process.env.LLM_TIMEOUT_MS ?? 60000),
    // 不重试：LangChain 默认会重试 6 次，5xx/超时会把一次运行拖到几分钟；
    // 两端都设为 0，行为才能一致（重试是上层策略，不是节点行为）
    maxRetries: 0,
    // 只有显式提供才传，避免把 undefined 传给 provider
    ...(typeof data.temperature === 'number' ? { temperature: data.temperature } : {}),
    ...(typeof data.maxTokens === 'number' ? { maxTokens: data.maxTokens } : {}),
  })

  let text = ''
  let usage: LlmStepResult['usage']
  try {
    // stream() 返回的是 Promise<AsyncIterable>，所以要 await 两次
    for await (const chunk of await client.stream(messages)) {
      // chunk.content 可能是 string / 内容块数组 / null
      const piece = stringifyContent(chunk.content)
      if (piece !== '') {
        text += piece
        emitToken(piece)
      }
      // usage 通常在最后一个 chunk 上（OpenAI 需要 stream_options.include_usage）
      const meta = chunk.usage_metadata
      if (meta && typeof meta.input_tokens === 'number') {
        usage = {
          promptTokens: meta.input_tokens,
          completionTokens: meta.output_tokens,
          totalTokens: meta.total_tokens,
        }
      }
    }
  } catch (error) {
    throw new RunError(mapLlmError(error))
  }

  return { text, model, usage }
}

/**
 * `llm` 节点入口：构造消息 → 调用 → 输出 `{ text, model, usage? }`。
 *
 * `usage` 用「有才加」而不是赋 `undefined`：`JSON.stringify` 会丢掉 undefined 键，
 * 但显式赋值会让 TS 类型与序列化意图不一致；这里保持输出对象干净。
 */
export async function runLlmNode(
  node: GraphNode,
  pool: Pool,
  emitToken: (text: string) => void,
): Promise<Record<string, unknown>> {
  const data = node.data as unknown as LlmNodeData
  const step = await runLlmStep(data, buildMessages(data, pool), emitToken)
  const output: Record<string, unknown> = { text: step.text, model: step.model }
  if (step.usage !== undefined) {
    output.usage = step.usage
  }
  return output
}

/** chunk.content 字符串化：字符串原样；数组拼各 text 部分；其余紧凑 JSON（与 api-py 对齐）。 */
function stringifyContent(content: MessageContent): string {
  if (typeof content === 'string') {
    return content
  }
  if (Array.isArray(content)) {
    return content
      .map((part) => {
        if (typeof part === 'string') {
          return part
        }
        const text = (part as unknown as { text?: unknown }).text
        return typeof text === 'string' ? text : ''
      })
      .join('')
  }
  if (content === null || content === undefined) {
    return ''
  }
  return JSON.stringify(content)
}

/** 异常 → 规范文案（完整规则见 03 篇）。 */
export function mapLlmError(error: unknown): string {
  const status = statusCodeOf(error)
  if (status !== null) {
    if (status === 401 || status === 403) {
      return 'llm request unauthorized'
    }
    if (status === 429) {
      return 'llm rate limited'
    }
    return `llm request failed: ${status}`
  }
  if (isTimeout(error)) {
    return 'llm request timed out'
  }
  return 'llm request failed: unknown'
}

/** 取异常的 HTTP 状态码（OpenAI SDK 的 APIError 带 `status`）。 */
function statusCodeOf(error: unknown): number | null {
  if (typeof error === 'object' && error !== null) {
    const status = (error as { status?: unknown }).status
    if (typeof status === 'number') {
      return status
    }
  }
  return null
}

/** 判定是否为超时（SDK 的 timeout 错误，或 message 含 timeout）。 */
function isTimeout(error: unknown): boolean {
  if (typeof error !== 'object' || error === null) {
    return false
  }
  const name = (error as { name?: unknown }).name
  if (name === 'APIConnectionTimeoutError' || name === 'TimeoutError' || name === 'AbortError') {
    return true
  }
  const message = (error as { message?: unknown }).message
  return typeof message === 'string' && /timed? ?out|timeout/i.test(message)
}
```

> **面试考点 · 为什么是 `for await (const chunk of await client.stream(...))`，两个 `await`？**
> LangChain 的 `stream()` 是 `async` 函数，返回的是 `Promise<AsyncIterable>`：
>
> ```text
> client.stream(messages)                    → Promise<AsyncIterable<Chunk>>
> await client.stream(messages)              → AsyncIterable<Chunk>
> for await (... of <AsyncIterable>)         → 逐个 chunk
> ```
>
> 少写外层 `await` 会得到 `Promise is not async iterable`。
> 对比 Python：`astream()` 返回的是 `AsyncIterator`，`async for` 直接迭代，**不要** `await`。两种语言的 API 形状不同，不能照抄。

> **面试考点 · `maxRetries: 0` 为什么必须两端一致？**
> LangChain 默认重试 6 次（指数退避）。假设 provider 限流：
>
> - 重试 6 次：一次运行可能卡 1~2 分钟才返回 `run_failed`；
> - 不重试：立刻返回 `llm rate limited`。
>
> 更麻烦的是**两端重试策略不同**（假设 TS 6 次、Python 0 次），同一个故障在 3001 上 90 秒失败、在 8000 上 1 秒失败——契约测试无法对齐。
> **重试属于“运行策略”，应该由上层（队列、用户重试按钮）决定，而不是藏在节点里。**

> **面试考点 · `chunk.content` 为什么可能是数组？**
> OpenAI 的流式响应里，纯文本通常给字符串；但多模态模型（或某些兼容实现）会给内容块数组：
>
> ```json
> [{"type": "text", "text": "你"}, {"type": "text", "text": "好"}]
> ```
>
> 直接 `text += chunk.content` 在 JS 里会得到 `[object Object]`。所以必须统一字符串化。
> 数组里只取 `text` 字段，其他类型（图片、音频）在纯文本节点里忽略——**不认识的块要跳过，而不是报错中断**。

> **面试考点 · `usage` 为什么在最后一个 chunk？**
> 流式协议里 token 数是**边生成边累计**的，只有生成结束才知道总数。OpenAI 会在 `stream_options.include_usage: true` 时，在最后一个 chunk 带 `usage`，且该 chunk 的 `choices` 通常为空。
> LangChain 会把它映射到 `chunk.usage_metadata`；部分兼容 provider 不返回，于是 `usage` 缺失——这正是契约把 `usage` 定义为可选字段的原因。
> **不要在前端计算 token 数**：中文按字符、英文按子词，估算永远不准。

> **面试考点 · 为什么 `temperature` 用展开运算符“有才传”，而不是直接传？**
> ```ts
> // 不好
> new ChatOpenAI({ temperature: data.temperature })   // undefined
> ```
> 大多数 SDK 会把 `undefined` 当成“未设置”，但有些实现会序列化成 `null`，provider 可能报 `temperature must be a number`。
> 展开写法让**键根本不存在**，语义最准确：
>
> ```ts
> ...(typeof data.temperature === 'number' ? { temperature: data.temperature } : {})
> ```
> 同理适用于 `maxTokens`。**可选配置的正确表达是“键不存在”，不是“键值为 undefined”。**

## 3. Python 实现

**完整文件路径：`apps/api-py/app/engine/nodes/llm.py`**（阶段 5 版本）

```python
"""`llm` 节点（阶段 5）：真实模型调用 + 流式 token。

与 api-ts 的 `llm.node.ts` 对齐：
- model 为空串时回退 `LLM_MODEL`；
- 每个非空 chunk 发一次 token；
- 输出 `{"text", "model", "usage"?}`；
- 错误经 `map_llm_error` 归一（完整规则见 03 篇）。
"""

import json
import os
from typing import Any, Awaitable, Callable

from langchain_core.messages import BaseMessage, HumanMessage, SystemMessage
from langchain_openai import ChatOpenAI

from app.engine.events import RunError
from app.engine.template import render_template


def build_messages(data: dict[str, Any], pool: dict[str, Any]) -> list[BaseMessage]:
    """`system` 非空才加；`prompt` 渲染后作为最后一条 `HumanMessage`。"""
    messages: list[BaseMessage] = []
    system = data.get("system")
    if isinstance(system, str) and system != "":
        messages.append(SystemMessage(content=system))
    messages.append(HumanMessage(content=render_template(str(data.get("prompt") or ""), pool)))
    return messages


async def run_llm_step(
    data: dict[str, Any],
    messages: list[BaseMessage],
    emit_token: Callable[[str], Awaitable[None]],
) -> dict[str, Any]:
    """执行一轮流式调用，返回 `{"text", "model", "usage"?}`。"""
    # 1. 没有密钥：立刻失败，不创建客户端、不发任何 token
    api_key = os.environ.get("LLM_API_KEY", "")
    if api_key == "":
        raise RunError("llm provider not configured: LLM_API_KEY")

    # 2. model 回退
    model = data.get("model") or os.environ.get("LLM_MODEL", "")

    # 3. 可选参数：有才传（等价于 TS 的展开写法）
    kwargs: dict[str, Any] = {}
    if isinstance(data.get("temperature"), (int, float)):
        kwargs["temperature"] = float(data["temperature"])
    if isinstance(data.get("maxTokens"), int):
        kwargs["max_tokens"] = data["maxTokens"]

    client = ChatOpenAI(
        api_key=api_key,
        model=model,
        base_url=os.environ.get("LLM_BASE_URL", "https://api.openai.com/v1"),
        # 环境变量是毫秒，OpenAI SDK 用秒
        timeout=float(os.environ.get("LLM_TIMEOUT_MS", "60000")) / 1000.0,
        # 不重试，与 TS 侧 maxRetries: 0 一致
        max_retries=0,
        **kwargs,
    )

    text = ""
    usage: dict[str, int] | None = None
    try:
        # astream 返回 AsyncIterator，直接 async for（不要 await）
        async for chunk in client.astream(messages):
            piece = _stringify_content(chunk.content)
            if piece != "":
                text += piece
                await emit_token(piece)
            meta = getattr(chunk, "usage_metadata", None)
            if isinstance(meta, dict) and isinstance(meta.get("input_tokens"), int):
                usage = {
                    "promptTokens": meta["input_tokens"],
                    "completionTokens": meta.get("output_tokens", 0),
                    "totalTokens": meta.get("total_tokens", 0),
                }
    except Exception as error:  # noqa: BLE001 - 统一映射为规范文案
        raise RunError(map_llm_error(error)) from error

    result: dict[str, Any] = {"text": text, "model": model}
    if usage is not None:
        result["usage"] = usage
    return result


async def run_llm_node(
    node: dict[str, Any],
    pool: dict[str, Any],
    emit_token: Callable[[str], Awaitable[None]],
) -> dict[str, Any]:
    """节点入口：构造消息 → 调用 → 输出契约形状。"""
    data: dict[str, Any] = node.get("data") or {}
    step = await run_llm_step(data, build_messages(data, pool), emit_token)
    output: dict[str, Any] = {"text": step["text"], "model": step["model"]}
    if "usage" in step:
        output["usage"] = step["usage"]
    return output


def _stringify_content(content: Any) -> str:
    """chunk.content 字符串化：字符串原样；列表拼 text 部分；其余紧凑 JSON。"""
    if isinstance(content, str):
        return content
    if isinstance(content, list):
        parts: list[str] = []
        for part in content:
            if isinstance(part, str):
                parts.append(part)
            elif isinstance(part, dict) and isinstance(part.get("text"), str):
                parts.append(part["text"])
        return "".join(parts)
    if content is None:
        return ""
    return json.dumps(content, ensure_ascii=False, separators=(",", ":"))


def map_llm_error(error: Exception) -> str:
    """异常 → 规范文案（完整规则见 03 篇）。"""
    status = _status_code(error)
    if status is not None:
        if status in (401, 403):
            return "llm request unauthorized"
        if status == 429:
            return "llm rate limited"
        return f"llm request failed: {status}"
    if _is_timeout(error):
        return "llm request timed out"
    return "llm request failed: unknown"


def _status_code(error: Exception) -> int | None:
    status = getattr(error, "status", None)
    return status if isinstance(status, int) else None


def _is_timeout(error: Exception) -> bool:
    name = type(error).__name__
    if name in {"APITimeoutError", "APIConnectionTimeoutError", "TimeoutError", "AbortError"}:
        return True
    return "timeout" in str(error).lower()
```

> **面试考点 · `timeout` 单位为什么两端不一样？**
> `LLM_TIMEOUT_MS` 是毫秒（Node 生态惯例）。`ChatOpenAI` 的 TS 版 `timeout` 收毫秒，Python 版（OpenAI SDK）收**秒**。
> 所以 Python 侧要 `/ 1000.0`。这类差异不会报错，只会“超时太早/太晚”，属于最难查的 bug。
> **任何跨语言配置项，都要在两侧注释里写清单位。**

> **面试考点 · 为什么 Python 的 `max_tokens` 和 TS 的 `maxTokens` 名字不同？**
> LangChain Python 的参数沿用了 OpenAI Python SDK 的 snake_case（`max_tokens`），而 TS 版用 camelCase（`maxTokens`）。
> 契约里的节点字段是 `maxTokens`（JSON 世界统一 camelCase），**只在构造客户端时做一次映射**——不要让 snake_case 泄漏到节点 `data` 里。
> 同理 `promptTokens` / `prompt_tokens`：契约输出统一 camelCase，Python 侧在赋值时转换。

> **面试考点 · 为什么 `text` 必须服务端累积，不能只让前端拼 token？**
> 三个原因：
>
> 1. 下游节点（`end` 模板、`knowledge` 查询）需要完整文本，它们拿不到前端状态；
> 2. `run_finished.outputs` 必须包含最终结果，客户端可能中途刷新页面；
> 3. token 帧可能丢（网络、代理），前端累积的结果会**静默偏差**。
>
> 前端拼接只用于“实时预览”；**权威结果永远以 `node_finished.output.text` 为准**。这也是为什么前端在 `run_finished` 后应该用输出覆盖预览（阶段 5 的 `04` 篇会做）。

## 5. 面试考点汇总

| 问题 | 答案要点 |
| --- | --- |
| 两个 `await` | `stream()` 返回 `Promise<AsyncIterable>`；Python `astream()` 直接是 AsyncIterator |
| `maxRetries: 0` | 重试是上层策略；两端策略不同会导致同一故障耗时不同 |
| `chunk.content` 三形态 | string / 内容块数组 / null；统一字符串化，未知块跳过 |
| `usage` 位置 | 最后一帧；provider 不返回则缺失，契约定义为可选 |
| 可选参数写法 | 键不存在优于值为 undefined；Python 用 `kwargs` |
| `text` 谁累积 | 服务端；前端拼接仅供预览，权威值在 `node_finished.output` |
| 超时单位 | Node 毫秒，OpenAI Python SDK 秒；跨语言配置必须注明单位 |
| 命名映射 | 契约 camelCase，SDK snake_case，只在构造客户端/输出时转换一次 |
| 无密钥失败时机 | 创建客户端之前抛 `RunError`，不发 token、不发 finished |
| 流式对比粒度 | 帧边界可不同；事件序列、完整文本、model 必须相同 |
